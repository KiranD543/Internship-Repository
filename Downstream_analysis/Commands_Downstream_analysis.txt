import os
from collections import defaultdict

# Load Prokka annotations: locus_tag → function
# And create mapping: strain → gene_callers_id → function
gene_functions = defaultdict(dict)

for strain_dir in os.listdir('prokka_output'):
    tsv_file = f'prokka_output/{strain_dir}/{strain_dir}.tsv'
    if not os.path.exists(tsv_file):
        continue

    locus_to_function = {}
    with open(tsv_file, 'r') as f:
        next(f)  # Skip header
        for line in f:
            parts = line.strip().split('\t')
            if len(parts) >= 2:
                locus_tag = parts[0].strip()
                function = parts[-1].strip() if parts[-1].strip() else 'hypothetical protein'
                locus_to_function[locus_tag] = function

    # Load external gene calls to map gene_callers_id → locus_tag
    ext_file = f'{strain_dir}_prokka-external-gene-calls.txt'
    if not os.path.exists(ext_file):
        continue

    with open(ext_file, 'r') as f:
        next(f)  # Skip header
        gene_id = 0
        for line in f:
            parts = line.strip().split('\t')
            if len(parts) >= 1:
                caller_id = parts[0].strip()
                # Find matching locus tag by position
                locus_tag = f'{list(locus_to_function.keys())[gene_id]}' if gene_id < len(locus_to_function) else None
                if locus_tag:
                    gene_functions[strain_dir][caller_id] = locus_to_function.get(locus_tag, 'hypothetical protein')
                gene_id += 1

print(f"Total strains with annotations: {len(gene_functions)}")

# Read unique genes and add Prokka annotations
sequences = defaultdict(list)
current_header = None
current_seq = ''
current_genome = ''

with open('Unique_genes/unique_genes_all.fasta', 'r') as f:
    for line in f:
        line = line.strip()
        if line.startswith('>'):
            # Save previous sequence
            if current_header and current_seq:
                sequences[current_genome].append((current_header, current_seq))

            # Extract genome name and gene caller ID
            current_genome = ''
            gene_caller_id = ''
            for part in line.split('|'):
                if 'genome_name_' in part:
                    current_genome = part.replace('genome_name_', '').strip()
                if 'gene_callers_id_' in part:
                    gene_caller_id = part.replace('gene_callers_id_', '').strip()

            # Get function from Prokka annotations
            function = gene_functions.get(current_genome, {}).get(gene_caller_id, 'hypothetical protein')
            current_header = f"{line}|function_{function}"
            current_seq = ''
        else:
            current_seq += line

    # Save last sequence
    if current_header and current_seq:
        sequences[current_genome].append((current_header, current_seq))

print(f"Total strains found in unique genes: {len(sequences)}")

# Split per strain and category
for genome, genes in sequences.items():
    hypothetical = []
    known = []
    all_genes = []

    for header, seq in genes:
        entry = f"{header}\n{seq}\n"
        all_genes.append(entry)

        if 'hypothetical protein' in header.lower():
            hypothetical.append(entry)
        else:
            known.append(entry)

    # Write output files
    with open(f'Unique_genes/All_unique_genes/all_{genome}.fasta', 'w') as f:
        f.writelines(all_genes)

    with open(f'Unique_genes/Hypothetical_proteins/hypothetical_{genome}.fasta', 'w') as f:
        f.writelines(hypothetical)

    with open(f'Unique_genes/Known_functions/known_{genome}.fasta', 'w') as f:
        f.writelines(known)

    print(f"{genome}:")
    print(f"  Total: {len(all_genes)} genes")
    print(f"  Hypothetical: {len(hypothetical)} genes")
    print(f"  Known functions: {len(known)} genes")

import os

# Read all unique gene headers and sequences
unique_genes = []

current_seq = ''

with open('Unique_genes/unique_genes_all_blast.fasta', 'r') as f:
    for line in f:
        line = line.strip()
        if line.startswith('>'):

            if current_header:

                gene_caller = ''

                for part in current_header.split('|'):

                    if 'gene_cluster_' in part:

                    'genome': genome,
                    'gene_cluster': gene_cluster,
                    'gene_callers_id': gene_caller,
                    'sequence': current_seq
                })
            current_header = line.lstrip('>')

        else:

    # Save last sequence

        gene_cluster = ''

            if 'gene_cluster_' in part:
                gene_cluster = part.replace('gene_cluster_', '')
        unique_genes.append({

            'genome': genome,
            'gene_cluster': gene_cluster,

        })

                              'sstart', 'send', 'evalue', 'bitscore', 'function'])

df_best = df_blast.sort_values('evalue').groupby('query').first().reset_index()

                                     'alignment_length', 'evalue',

                            on='query', how='left')

# Fill unknown for genes without BLAST hit
df_merged['subject'] = df_merged['subject'].fillna('Unknown')
df_merged['function'] = df_merged['function'].fillna('Unknown - no BLAST hit found')
df_merged['pident'] = df_merged['pident'].fillna(0)
df_merged['evalue'] = df_merged['evalue'].fillna('-')
df_merged['bitscore'] = df_merged['bitscore'].fillna(0)
df_merged['alignment_length'] = df_merged['alignment_length'].fillna(0)

df_no_hits = df_merged[df_merged['subject'] == 'Unknown']

# Save all results combined
df_merged.to_csv('BLAST_results/All_BLAST_results/all_strains_blast.csv', index=False)
df_hits.to_csv('BLAST_results/BLAST_identified/all_strains_identified.csv', index=False)

# Save per strain CSV

    # All results

import os
from collections import defaultdict

# GABA operon genes
gaba_genes = {
	'gadB': ['glutamate decarboxylase', 'gadB'],
	'gadC': ['gadC', 'glutamate/GABA antiporter', 'amino acid antiporter'],
	'gabD': ['gabD', 'succinic semialdehyde dehydrogenase'],
	'gltT': ['gltT', 'glutamate transporter', 'sodium/glutamate symporter']	
}

os.makedirs('Downstream_analysis/GABA', exist_ok=True)

# Results per strain
results = {}

for kegg_file in sorted(os.listdir('Downstream_analysis/Databases')):
	if not kegg_file.endswith('_kegg_unique.txt'):
		continue
	
	strain = kegg_file.replace('_kegg_unique.txt', '')
	strain_results = {gene: [] for gene in gaba_genes}
	
	with open(f'Downstream_analysis/Databases/{kegg_file}', 'r') as f:
		next(f) #skip header
		for line in f:
			parts = line.strip().split('\t')
			if len(parts) < 4:
				continue
			gene_id = parts[0]
			function = parts[3].lower()
			
			# Check for each GABA gene
			for gene_name, keywords in gaba_genes.items():
				for keyword in keywords:
					if keyword.lower() in function:
						strain_results[gene_name].append({
							'gene_id': gene_id,
							'function': parts[3],
							'evalue': parts[4] if len (parts) > 4 else 'N/A'
						})
						break
					
	results[strain] = strain_results

# Write detailed results per strain
for strain, strain_results in results.items():
	output_file = f'Downstream_analysis/GABA/{strain}_GABA.txt'
	with open(output_file, 'w') as f:
		f.write(f"=== GABA operon analysis: {strain} ===\n\n")
		
		# Operon completeness
		total_genes = len(gaba_genes)
		present_genes = sum(1 for hits in strain_results.values() if hits)
		completeness = (present_genes / total_genes) * 100
		
		f.write(f"Operon completeness: {present_genes}/{total_genes}/ genes ({completeness:.1f}%\n\n")
		
		# Details per gene
		for gene_name, hits in strain_results.items():
			f.write(f"--- {gene_name} ---\n")
			if hits:
				for hit in hits:
					f.write(f" Gene ID: {hit['gene_id']}\n")
					f.write(f" Function: {hit['function']}\n")
					f.write(f" E-value: {hit['evalue']}\n")
			else:
				f.write(f" Not found in unique genes\n")
			f.write("\n")
	
	print(f"{strain}: {present_genes}/{total_genes} GABA genes found ({completeness:.1f}%)")
	
# Write summary of all strains
summary_file = 'Downstream_analysis/GABA/GABA_summary_all_strains.txt'
with open(summary_file, 'w') as f:
	f.write("=== GABA operon summary - Lp. plantarum ===\n\n")
	f.write(f"{'Strain':<40} {'gadB':<8} {'gadC':<8} {'gabD':<8} {'gltT':<8} {'Complete':<10}\n")
	f.write("-" * 90 + "\n")
	
	for strain, strain_results in sorted(results.items()):
		gadB = "yes" if strain_results['gadB'] else "no"
		gadC = "yes" if strain_results['gadC'] else "no"
		gabD = "yes" if strain_results['gabD'] else "no"
		gltT = "yes" if strain_results['gltT'] else "no"
		present = sum(1 for hits in strain_results.values() if hits)
		complete = "Full" if present == 4 else f"{present}/4"
		f.write(f"{strain:<40} {gadB:<8} {gadC:<8} {gabD:8} {gltT:<8} {complete:<10}\n")
		
print(f"\nSummary saved to: {summary_file}")

# Load Prokka locus tag => gene caller ID mapping
locus_to_callerid = defaultdict(dict)
for strain in unique_per_strain.keys():
	ext_file = f'{strain}_prokka-external-gene-calls.txt'
	tsv_file = f'prokka_output/{strain}/{strain}.tsv'
	
	if not os.path.exists(ext_file) or not os.path.exists(tsv_file):
		continue
	
	# Get locus tags in order from TSV
	locus_tags = []
	with open(tsv_file, 'r') as f:
		next(f)
		for line in f:
			parts = line.strip().split('\t')
			if len(parts) >= 1:
				locus_tags.append(parts[0].strip())
	
	# Map gene caller ID to locus tag by position
	with open(ext_file, 'r') as f:
		next(f)
		for i, line in enumerate(f):
			parts = line.strip().split('\t')
			if len(parts) >= 1 and i < len(locus_tags):
				caller_id = parts[0].strip()
				locus_tag = locus_tags[i]
				locus_to_callerid[strain][locus_tag] = caller_id

	region_files = set(glob.glob(f'{antismash_dir}/*.region*.gbk'))
	
	seen_regions = set()
	strain_results = []

print(f"\nSummary saved to: {summary_file}")
