To find the amount of unique genes per strain:
$ python3 << 'EOF'
from collections import defaultdict

counts = defaultdict(int)

with open('unique_genes_all.fasta', 'r') as f:
    for line in f:
        if line.startswith('>'):
            # Get the genome name from the header
            parts = line.strip().split('|')
            for part in parts:
                if 'genome_name:' in part:
                    genome = part.replace('genome_name:', '').strip()
                    counts[genome] += 1

for genome, count in sorted(counts.items()):
    print(f"{genome}: {count} unique genes")


To make a fasta of the unique genes:
$ python3 << 'EOF'
from collections import defaultdict

# Read all sequences
sequences = {}
current_header = None
current_genome = None

with open('unique_genes_all.fasta', 'r') as f:
    for line in f:
        line = line.strip()
        if line.startswith('>'):
            current_header = line
            # Extract genome name from header
            for part in line.split('|'):
                if 'genome_name:' in part:
                    current_genome = part.replace('genome_name:', '').strip()
            sequences[current_header] = {'genome': current_genome, 'seq': ''}
        else:
            sequences[current_header]['seq'] += line

# Group by strain
genomes = defaultdict(list)
for header, data in sequences.items():
    genomes[data['genome']].append((header, data['seq']))

# Write separate FASTA file per strain
for genome, genes in genomes.items():
    filename = f"unique_genes_{genome}.fasta"
    with open(filename, 'w') as f:
        for header, seq in genes:
            f.write(f"{header}\n{seq}\n")
    print(f"{genome}: {len(genes)} unique genes → {filename}")

EOF
