To annotate with prokka:
$ for fasta in FastaFiles/*.fasta; do
  strain=$(basename $fasta .fasta)
  echo "Annotating: $strain"
  prokka --outdir prokka_output/$strain \
  --prefix $strain \
  --genus Lactiplantibacillus \
  --species plantarum \
  --kingdom Bacteria \
  --usegenus \
  --cpus 8 \
  $fasta
done

To make a combined summary file: 
$ for txt in prokka_output/*/*.txt; do
  strain=$(basename $txt .txt)
  contigs=$(grep "contigs:" $txt | awk '{print $2}')
  bases=$(grep "bases:" $txt | awk '{print $2}')
  cds=$(grep "CDS:" $txt | awk '{print $2}')
  echo "$strain: contigs=$contigs, bases=$bases, CDS=$cds"
  echo ""
done > prokka_summary.txt

To reformat the fasta files:
$ for fasta in FastaFiles/*.fasta; do

To create the contigs databases:
$ for fasta in FastaFiles_reformatted/*_reformatted.fasta; do
  strain=$(basename $fasta _reformatted.fasta)
  echo "Processing: $strain"
  anvi-gen-contigs-database \
  -f $fasta \
  -o ${strain}.db \
  -n $strain
done

To create a summary comparable to Roary and Panaroo:
python3 << 'EOF'
import pandas as pd

# Load the file
df = pd.read_csv('pangenome_summary/Lp_plantarum_gene_clusters_summary.txt',
                 sep='\t',
                 usecols=['gene_cluster_id', 'num_genomes_gene_cluster_has_hits'])

# Delete duplicates
df = df.drop_duplicates(subset='gene_cluster_id')

total_genomes = 42

# Calculate the percentage presence
df['percentage'] = df['num_genomes_gene_cluster_has_hits'] / total_genomes * 100

# Categorize like Roary and Panaroo
core = len(df[df['percentage'] >= 99])
soft_core = len(df[(df['percentage'] >= 95) & (df['percentage'] < 99)])
shell = len(df[(df['percentage'] >= 15) & (df['percentage'] < 95)])
cloud = len(df[df['percentage'] < 15])
total = len(df)

# Write to file
with open('anvio_pangenome_summary.txt', 'w') as f:
    f.write("=== Pangenome Summary (Anvi'o) ===\n\n")
    f.write(f"Core genes    (99% <= strains <= 100%)  {core}\n\n")
    f.write(f"Soft core     (95% <= strains < 99%)    {soft_core}\n\n")
    f.write(f"Shell genes   (15% <= strains < 95%)    {shell}\n\n")
    f.write(f"Cloud genes   (0%  <= strains < 15%)    {cloud}\n\n")
    f.write(f"Total genes   (0%  <= strains <= 100%)  {total}\n")

print("Summary saved to anvio_pangenome_summary.txt")
EOF


To extract the unique genes:
$ anvi-get-sequences-for-gene-clusters \
  -p pangenome_output/Lp_plantarum-PAN.db \
  -g Lp_plantarum-GENOMES.db \
  --max-num-genomes-gene-cluster-occurs 1 \
  --min-num-genomes-gene-cluster-occurs 1 \

To put them in a file with a fasta header that doesn't have any ":":
$ sed 's/:/\_/g' unique_genes_all_raw.fasta > unique_genes_all.fasta 


To see the amount of unique genes per strain:
python3 << 'EOF'
from collections import defaultdict

counts = defaultdict(int)

with open('unique_genes_all.fasta', 'r') as f:
    for line in f:
        if line.startswith('>'):
            parts = line.strip().split('|')
            for part in parts:
                if 'genome_name_' in part:
                    genome = part.replace('genome_name_', '').strip()
                    counts[genome] += 1

for genome, count in sorted(counts.items()):
    print(f"{genome}: {count} unique genes")
EOF



To make FASTA files for the unique genes of each strain:
python3 << 'EOF'
from collections import defaultdict

# Read all sequences
sequences = {}
current_header = None
current_genome = None

with open('unique_genes_all.fasta', 'r') as f:
    for line in f:
        line = line.strip()
        if line.startswith('>'):
            current_header = line
            for part in line.split('|'):
                if 'genome_name_' in part:
                    current_genome = part.replace('genome_name_', '').strip()
            sequences[current_header] = {'genome': current_genome, 'seq': ''}
        else:
            sequences[current_header]['seq'] += line

# Group by strain
genomes = defaultdict(list)
for header, data in sequences.items():
    genomes[data['genome']].append((header, data['seq']))

# Write separate FASTA file per strain in Unique_genes directory
for genome, genes in genomes.items():
    filename = f"Unique_genes/unique_genes_{genome}.fasta"
    with open(filename, 'w') as f:
        for header, seq in genes:
            f.write(f"{header}\n{seq}\n")
    print(f"{genome}: {len(genes)} unique genes → {filename}")
EOF
